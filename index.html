Sample Automated Test (JUnit 5)

@Test
@DisplayName("Should throw ValidationException when user email format is invalid")
void testInvalidEmailValidation() {
    User invalidUser = new User("Salah Reda", "invalid-email-format", "password123");
    
    ValidationException exception = assertThrows(
        ValidationException.class, 
        () -> userService.registerUser(invalidUser),
        "Expected registerUser() to throw, but it didn't"
    );

    assertEquals("Invalid email format provided.", exception.getMessage());
}
